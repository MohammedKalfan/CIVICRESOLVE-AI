CATEGORIZATION (keyword-weighted classifier + user hint) # ---------------------------------------------------------------------- CATEGORIES = { "Waterlogging": { "waterlog": 3, "waterlogging": 3, "flooded": 3, "flood": 2, "stagnant": 2, "puddle": 2, "rainwater": 2, "drain water": 2, "sewage overflow": 2, "water logged": 3, }, "Road Damage": { "pothole": 3, "potholes": 3, "crack": 2, "cracks": 2, "broken road": 3, "asphalt": 1, "speed breaker": 2, "road cave": 3, "uneven road": 2, "road damage": 3, "paved": 1, }, "Waste / Garbage": { "garbage": 3, "trash": 3, "waste": 2, "dustbin": 3, "bin overflow": 3, "dump": 2, "dumping": 2, "litter": 2, "debris": 2, "overflowing bin": 3, "rotting": 2, "stink": 2, }, "Streetlight": { "streetlight": 3, "street light": 3, "lamp post": 3, "pole light": 3, "no light": 3, "dark street": 3, "flickering light": 2, "light pole": 2, }, "Water Supply": { "no water": 3, "water supply": 3, "low pressure": 2, "pipe burst": 3, "leaking pipe": 3, "brown water": 2, "contaminated water": 3, "dry tap": 3, "water tanker": 1, "pipeline": 2, }, "Public Safety": { "live wire": 3, "open wire": 3, "electric shock": 3, "transformer": 2, "fallen tree": 3, "accident prone": 3, "open drain": 2, "drowning": 3, "fire hazard": 3, "broken railing": 2, "unsafe": 2, }, } USER_HINT_BONUS = 2.5 # if user picked a category manually, trust it a bit def _tokens(text): return set(re.findall(r"[a-z0-9']+", text.lower())) def categorize(text, user_hint=None): """Return (category, confidence, top_reasons). Keyword-weighted scoring.""" t = text.lower() best, best_score = "Other", 0.0 scores = {} for cat, kw in CATEGORIES.items(): s = 0.0 hits = [] for k, w in kw.items(): if k in t: s += w hits.append(k) if user_hint == cat: s += USER_HINT_BONUS scores[cat] = s if s > best_score: best, best_score = cat, s conf = round(min(best_score / 6.0, 1.0), 2) if best_score else 0.0 reasons = [f"matched keywords: {', '.join(sorted(k for k in CATEGORIES.get(best, {}) if k in t))[:80]}"] if best != "Other" else ["no strong keyword match"] if user_hint == best: reasons.append("agrees with the category you selected") return best, conf, reasons # ---------------------------------------------------------------------- # 2. SUMMARIZATION (extractive — first key sentence + location) # ---------------------------------------------------------------------- def summarize(text, category, location): """Extractive summary: most informative sentence, trimmed to ~150 chars.""" sentences = [s.strip() for s in re.split(r"(?<=[.!?])\s+", text.strip()) if s.strip()] if not sentences: return f"{category} issue reported near {location}." # score sentences: prefer ones with severity keywords and early position hot = ("urgent", "danger", "injur", "child", "night", "accident", "blocked", "overflow", "broken", "leak", "live wire", "dangling", "burst", "dead", "no water") best, bs = sentences[0], -1 for i, s in enumerate(sentences[:4]): sc = sum(1 for w in hot if w in s.lower()) * 2 - i if sc > bs: best, bs = s, sc summary = best[:157] + ("..." if len(best) > 157 else "") return f"[{category}] {summary} (near {location})" # ---------------------------------------------------------------------- # 3. DUPLICATE DETECTION (TF-IDF cosine + geographic proximity) # ---------------------------------------------------------------------- def _tfidf_matrix(texts): """Tiny hand-rolled TF-IDF -> (matrix, vocab). No sklearn needed.""" docs = [re.findall(r"[a-z0-9']+", t.lower()) for t in texts] df = {} for d in docs: for w in set(d): df[w] = df.get(w, 0) + 1 N = len(docs) idf = {w: math.log((N + 1) / (c + 1)) + 1 for w, c in df.items()} vocab = sorted(idf) idx = {w: i for i, w in enumerate(vocab)} M = np.zeros((N, len(vocab))) for r, d in enumerate(docs): for w in d: M[r, idx[w]] += 1.0 if M[r].sum(): M[r] /= M[r].sum() M *= np.array([idf[w] for w in vocab]) norms = np.linalg.norm(M, axis=1, keepdims=True) norms[norms == 0] = 1 return M / norms def haversine_m(lat1, lon1, lat2, lon2): R = 6371000 p1, p2 = math.radians(lat1), math.radians(lat2) dp, dl = math.radians(lat2 - lat1), math.radians(lon2 - lon1) a = math.sin(dp / 2) ** 2 + math.cos(p1) * math.cos(p2) * math.sin(dl / 2) ** 2 return 2 * R * math.asin(math.sqrt(a)) def find_duplicates(df, text_sim=0.45, max_dist_m=250.0): """Union-find: reports that are textually AND geographically near are one issue. Returns (parent_map, group_map).""" texts = (df["text"].astype(str) + " " + df["location"].astype(str)).tolist() M = _tfidf_matrix(texts) if len(df) else np.zeros((0, 0)) parent = list(range(len(df))) def find(a): while parent[a] != a: parent[a] = parent[parent[a]] a = parent[a] return a def union(a, b): ra, rb = find(a), find(b) if ra != rb: parent[max(ra, rb)] = min(ra, rb) for i in range(len(df)): for j in range(i + 1, len(df)): sim = float(M[i] @ M[j]) dist = haversine_m(df.iloc[i].lat, df.iloc[i].lon, df.iloc[j].lat, df.iloc[j].lon) if sim >= text_sim and dist <= max_dist_m: union(i, j) groups = {} for i in range(len(df)): groups.setdefault(find(i), []).append(i) return parent, groups # ---------------------------------------------------------------------- # 4. EXPLAINABLE PRIORITIZATION (additive scoring, every point has a reason) # ---------------------------------------------------------------------- CAT_BASE = {"Public Safety": 40, "Waterlogging": 26, "Water Supply": 24, "Road Damage": 22, "Streetlight": 18, "Waste / Garbage": 14, "Other": 10} SEVERITY = { 25: ["live wire", "electric shock", "fire hazard", "drowning", "injur", "collapse", "child", "school", "hospital", "accident"], 15: ["night", "dark", "blocked", "overflow", "dengue", "malaria", "smell", "rotting", "elderly", "no water", "fallen"], 8: ["urgent", "danger", "unsafe", "heavy", "three days", "week"], } def score_priority(text, category, dup_count, age_days, in_hotspot): """Returns (score 0-100, band, list_of_reason_strings).""" t = text.lower() score, reasons = 0.0, [] base = CAT_BASE.get(category, 10) score += base reasons.append(f"+{base} base weight for '{category}' category") for pts, words in SEVERITY.items(): hits = [w for w in words if w in t] if hits: score += pts reasons.append(f"+{pts} severity: mentions {', '.join(hits[:3])}") break if dup_count > 1: pts = min((dup_count - 1) * 8, 24) score += pts reasons.append(f"+{pts} impact: {dup_count} reports describe the same issue") if age_days <= 2: score += 8 reasons.append("+8 fresh report (raised in last 48h)") elif age_days <= 7: score += 4 reasons.append("+4 recent report (within a week)") else: score -= 6 reasons.append("-6 stale report (>7 days old)") if in_hotspot: score += 10 reasons.append("+10 located inside a detected hotspot cluster") score = int(round(max(0, min(100, score)))) band = ("Critical" if score >= 70 else "High" if score >= 50 else "Medium" if score >= 32 else "Low") return score, band, reasons # ---------------------------------------------------------------------- # 5. HOTSPOT / CLUSTER DETECTION (greedy density clustering in meters) # ---------------------------------------------------------------------- def find_hotspots(df, radius_m=350.0, min_reports=2): """Greedy geo-cluster; a cluster with >=min_reports becomes a hotspot.""" pts = df[["lat", "lon"]].values used = np.zeros(len(pts), bool) clusters = [] for i in range(len(pts)): if used[i]: continue d = np.array([haversine_m(pts[i][0], pts[i][1], p[0], p[1]) for p in pts]) members = np.where((d <= radius_m) & (~used))[0] if len(members) >= min_reports: used[members] = True clusters.append(members.tolist()) return clusters def action_brief(cluster, df, now=None): """Turn a cluster of reports into a <1-minute decision brief.""" sub = df.iloc[cluster] cats = sub["category"].value_counts() top_cat = cats.index[0] n = len(sub) center = (sub.lat.mean(), sub.lon.mean()) loc = sub["location"].mode().iloc[0] if not sub["location"].mode().empty else "the area" days = int(max(1, (pd_max(sub.created_at) - pd_min(sub.created_at)).days) + 1) if n > 1 else 1 priority = sub["priority_score"].max() brief = (f"**{top_cat} hotspot — {loc}.** {n} related reports within ~350m, " f"spanning ~{days} day(s). Dominant issue: {top_cat.lower()} " f"({cats.iloc[0]} of {n} reports" + (f", co-occurring with {', '.join(cats.index[1:3])}" if len(cats) > 1 else "") + f"). Highest priority in cluster: {int(priority)}/100. " f"**Suggested action:** dispatch a joint crew to inspect {loc} " f"({center[0]:.5f}, {center[1]:.5f}) and fix root causes together rather than " f"closing tickets one by one.") return brief, top_cat, n, center def pd_max(s): # tiny helpers so we don't import pandas here return max(s) def pd_min(s): return min(s) # ---------------------------------------------------------------------- # 6. FULL PIPELINE over the whole dataset # ---------------------------------------------------------------------- def run_pipeline(df, now=None): """Enriches a reports dataframe with AI outputs. Returns enriched copy.""" now = now or datetime.now() df = df.copy() _, groups = find_duplicates(df) dup_of = {} dup_size = {} for members in groups.values(): if len(members) > 1: lead = members[0] for m in members: dup_of[m] = lead dup_size[m] = len(members) # hotspots on unresolved reports open_mask = ~df["status"].isin(["Resolved"]) open_idx = df.index[open_mask] clusters = find_hotspots(df.loc[open_idx]) hotspot_ids = set() for c in clusters: for local_i in c: hotspot_ids.add(open_idx[local_i]) out = [] for pos, row in enumerate(df.itertuples()): age = (now - row.created_at).total_seconds() / 86400.0 dup_count = dup_size.get(pos, 1) score, band, reasons = score_priority(row.text, row.category, dup_count, age, row.Index in hotspot_ids) out.append({ "ai_category": row.category, "summary": summarize(row.text, row.category, row.location), "dup_group": dup_of.get(pos, pos), "dup_count": dup_count, "priority_score": score, "priority_band": band, "priority_reasons": " | ".join(reasons), "in_hotspot": row.Index in hotspot_ids, }) enrich = __import__("pandas").DataFrame(out, index=df.index) df = __import__("pandas").concat([df, enrich], axis=1) briefs = [] for c in clusters: global_idx = [open_idx[i] for i in c] brief, top_cat, n, center = action_brief(global_idx, df) briefs.append({"brief": brief, "category": top_cat, "reports": n, "center": center}) return df, briefs # ==================== SEED DATA (data.py) ==================== CITY = (22.7196, 75.8577) # Indore, as sample city # (text, category, location_name, lat, lon, days_ago, status, assignee) BASE = [ # ---------- URGENT SAFETY ---------- ("Live electric wire dangling from a pole right outside Saraswati School gate.
